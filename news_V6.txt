        if (rest.indexOf('c/') === 0) cat = rest.slice(2);      // старый вид /c/football
        else if (findCat(rest)) cat = rest;                      // /ru/news/football — раздел
        else if (rest) news = rest;                               // /ru/news/<адрес новости>
      }
      if (!news && h.indexOf('n/') === 0) news = h.slice(2);
      if (!news && !cat && h.indexOf('c/') === 0) cat = h.slice(2);
      if (news) return {page: 'news', id: news};
      return {page: 'list', cat: findCat(cat)};
    }

    var tags = {}, origTitle = null;
    function tag(key, selector, make, attr, value) {
      var t = tags[key];
      if (!t) {
        var e = document.head.querySelector(selector), created = !e;
        if (created) { e = make(); document.head.appendChild(e); }
        t = tags[key] = {el: e, created: created, attr: attr, orig: created ? null : e.getAttribute(attr)};
      }
      t.el.setAttribute(attr, value);
    }
    function meta(name, value, prop) {
      var a = prop ? 'property' : 'name';
      tag(a + ':' + name, 'meta[' + a + '="' + name + '"]', function () {
        var m = document.createElement('meta'); m.setAttribute(a, name); return m;
      }, 'content', value);
    }
    function dropTag(key) {   // вернуть как было до нас
      var t = tags[key];
      if (!t) return;
      if (t.created) t.el.remove(); else if (t.orig != null) t.el.setAttribute(t.attr, t.orig);
      delete tags[key];
    }
    function dropLd() { var s = document.getElementById('mbn-ld'); if (s) s.remove(); }

    function seo(o) {   // ошибка в SEO-части не должна ломать показ новостей
      try { seoApply(o); } catch (e) {}
    }
    function seoApply(o) {
      if (origTitle === null) origTitle = document.title;
      document.title = o.title;
      meta('description', o.desc);
      tag('canonical', 'link[rel="canonical"]', function () {
        var l = document.createElement('link'); l.rel = 'canonical'; return l;
      }, 'href', o.url);
      meta('og:title', o.title, true);
      meta('og:description', o.desc, true);
      meta('og:url', o.url, true);
      meta('og:type', o.type, true);
      meta('og:site_name', CONFIG.siteName, true);
      if (o.image) meta('og:image', o.image, true); else dropTag('property:og:image');
      if (o.notFound) { meta('robots', 'noindex'); meta('prerender-status-code', '404'); }
      else { dropTag('name:robots'); dropTag('name:prerender-status-code'); }
      dropLd();
      if (o.ld) {
        var s = document.createElement('script');
        s.type = 'application/ld+json'; s.id = 'mbn-ld';
        s.textContent = JSON.stringify(o.ld).replace(/</g, '\\u003c');
        document.head.appendChild(s);
      }
    }

    function when(iso, full) {
      var d = new Date(iso);
      var t = d.toLocaleTimeString('ru-RU', {hour: '2-digit', minute: '2-digit'});
      if (full) return d.toLocaleDateString('ru-RU', {day: 'numeric', month: 'long', year: 'numeric'}) + ', ' + t;
      var a = new Date(d), b = new Date();
      a.setHours(0, 0, 0, 0); b.setHours(0, 0, 0, 0);
      var days = Math.round((b - a) / 864e5);
      if (days === 0) return 'Сегодня, ' + t;
      if (days === 1) return 'Вчера, ' + t;
      return d.toLocaleDateString('ru-RU', {day: 'numeric', month: 'long'}) + ', ' + t;
    }

    // высота закреплённой шапки сайта (если она есть), чтобы она не закрывала начало новости
    function fixedTop() {
      var w = window.innerWidth || document.documentElement.clientWidth, h = 0, seen = [];
      var cand = [];
      [0.2, 0.5, 0.8].forEach(function (x) {
        if (document.elementsFromPoint && w) cand = cand.concat(document.elementsFromPoint(w * x, 2));
      });
      cand = cand.concat([].slice.call(document.querySelectorAll('header, [class*="header"], [class*="Header"]')));
      cand.forEach(function (e) {
        for (; e && e !== document.body && e !== document.documentElement; e = e.parentElement) {
          if (seen.indexOf(e) >= 0 || root.contains(e)) return;
          seen.push(e);
          var pos = getComputedStyle(e).position, r = e.getBoundingClientRect();
          if ((pos === 'fixed' || pos === 'sticky') && r.top <= 1 && r.bottom > 0 && r.height < window.innerHeight * 0.4) {
            h = Math.max(h, r.bottom); return;
          }
        }
      });
      return h;
    }
    // поднять экран к началу блока новостей — только если пользователь уже пролистал ниже него
    function upToTop() {
      var top = root.getBoundingClientRect().top - fixedTop();
      if (top < 0) window.scrollTo(0, Math.max(0, window.scrollY + top));
    }

    function render(isNav) {
      var r = route();
      var key = r.page + '/' + (r.id || (r.cat ? r.cat.slug : ''));
      if (isNav && lastKey.indexOf('list/') === 0) listScroll = window.scrollY;
      if (r.page === 'news') {
        drawNews(r.id);
        if (isNav) upToTop();
      } else {
        seq++;
        if (key !== lastKey && lastKey.indexOf('list/') === 0) shown = PAGE;   // сменили раздел
        drawList(r.cat);
        if (isNav) {
          if (lastKey.indexOf('news/') === 0 && listScroll) window.scrollTo(0, listScroll);   // назад из новости — на прежнее место
          // переключение разделов: экран не трогаем
        }
      }
      lastKey = key;
      lastHref = location.href;
    }

    function onUrlChange() { if (location.href !== lastHref) { virtual = null; render(true); } }

    function onClick(e) {
      if (e.defaultPrevented || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
      var a = e.target.closest ? e.target.closest('a[data-nav]') : null;
      if (!a || !root.contains(a)) return;
      e.preventDefault();
      if (CONFIG.urlMode === 'hash') {
        var target = a.getAttribute('href'), before = location.href;
        try { location.hash = target; } catch (err) {}
        // окружение не даёт менять адрес (например, предпросмотр файла) — переходим «внутри» страницы
        if (location.href === before) { virtual = target; render(true); }
        return;
      }
      try { history.pushState(null, '', a.getAttribute('href')); }
      catch (err) { location.href = a.href; return; }
      onUrlChange();
    }

    function loadBanners() {
      return fetch(BASE + 'banners.json?t=' + Date.now())
        .then(function (r) { return r.ok ? r.json() : {}; })
        .then(function (b) { banners = b && typeof b === 'object' ? b : {}; })
        .catch(function () { banners = {}; });
    }

    function load() {
      fetch(BASE + 'news.json?t=' + Date.now())
        .then(function (r) { if (!r.ok) throw new Error(r.status); return r.json(); })
        .then(function (data) {
          loaded = true;
          items = (data.items || []).filter(function (n) { return n.category; });
          upd.className = 'foot';
          upd.textContent = data.updated_at ? 'Лента обновлена: ' + when(data.updated_at) : '';
          var r = route();
          if (r.page === 'list') drawList(r.cat);
          else {   // на странице статьи экран не дёргаем — только обновляем разделы и «Другие новости»
            drawHeader(false, artCat);
            var box = view.querySelector('.art');
            if (box && box.querySelector('.body')) { fillOthers(box, r.id); }
          }
        })
        .catch(function (e) {
          upd.className = 'foot err';
          upd.textContent = 'Не удалось загрузить новости (' + e.message + ')';
          if (!loaded && route().page === 'list') { view.innerHTML = ''; ready(); }
        });
    }

    window.addEventListener('popstate', onUrlChange);
    window.addEventListener('hashchange', onUrlChange);
    var timer = setInterval(load, 120000);
    // ушли с лендинга на другую страницу сайта — убираем за собой title, canonical и микроразметку
    var guard = setInterval(function () { if (!document.documentElement.contains(root)) window.__mbnStop(); }, 1500);

    window.__mbnStop = function () {
      clearInterval(timer); clearInterval(guard);
      root.removeEventListener('click', onClick);
      window.removeEventListener('popstate', onUrlChange);
      window.removeEventListener('hashchange', onUrlChange);
      Object.keys(tags).forEach(dropTag);
      dropLd();
      if (origTitle !== null) document.title = origTitle;
      window.__mbnStop = null;
    };

    loadBanners().then(function () {
      var top = document.getElementById('mbn-adtop');
      if (top) { top.innerHTML = ''; addAd(top, 'top'); }   // длинный баннер сверху, под меню разделов
      render(false); load();
    });
