  <script>
  (function () {
    /* ================== НАСТРОЙКИ (единственное, что может понадобиться менять) ==================
       urlMode — как выглядят адреса новостей:
         'query' → …/melbetpremiumsize?news=sbornaya-rossii-obygrala-iran   (работает без доработок платформы)
         'path'  → /ru/news/sbornaya-rossii-obygrala-iran                   (нужно, чтобы платформа отдавала этот
                                                                             лендинг по всем адресам pathBase + *)
         'hash'  → …/melbetpremiumsize#n/sbornaya-…                          (как было раньше; НЕ индексируется)
       Если меняете urlMode/pathBase — поменяйте и SITE_*_URL в generate.py (адреса в sitemap.xml). */
    var CONFIG = {
      urlMode: 'query',
      pathBase: '/ru/news/',
      author: 'Редакция Melbet',
      siteName: 'Melbet'
    };
    /* ============================================================================================ */

    // файл открыт с диска (превью), а не с сайта — адреса с ? там не работают, используем #
    if (!/^https?:$/.test(location.protocol)) CONFIG.urlMode = 'hash';

    var BASE = 'https://raw.githubusercontent.com/anastasiyamakarenko/melbet-news/main/data/';
    // раздел: название в данных, кусок адреса, заголовок h1 страницы раздела
    var CATS = [
      ['Футбол', 'football', 'Новости футбола'], ['Хоккей', 'hockey', 'Новости хоккея'],
      ['Баскетбол', 'basketball', 'Новости баскетбола'], ['Теннис', 'tennis', 'Новости тенниса'],
      ['Фигурное катание', 'figure-skating', 'Новости фигурного катания'], ['Волейбол', 'volleyball', 'Новости волейбола'],
      ['Биатлон', 'biathlon', 'Новости биатлона'], ['Единоборства', 'martial-arts', 'Новости единоборств'],
      ['Автоспорт', 'motorsport', 'Новости автоспорта'], ['Киберспорт', 'esports', 'Новости киберспорта'],
      ['Другое', 'other', 'Другие новости спорта']
    ].map(function (c) { return {label: c[0], slug: c[1], h1: c[2]}; });
    var HOME_TITLE = 'Новости спорта';
    var HOME_DESC = 'Свежие новости спорта: футбол, хоккей, баскетбол, теннис и другие виды. Результаты матчей, трансферы, заявления игроков и тренеров.';
    var PAGE = 20;

    // если платформа перерисовала страницу и скрипт запустился второй раз — выключаем прошлый экземпляр
    if (window.__mbnStop) { try { window.__mbnStop(); } catch (e) {} }
    window.prerenderReady = false;   // сигнал пререндеру: страница ещё не готова

    var root = document.getElementById('mbn');
    var logo = document.getElementById('mbn-logo');
    var nav = document.getElementById('mbn-nav');
    var view = document.getElementById('mbn-view');
    var upd = document.getElementById('mbn-upd');
    var items = [], loaded = false, shown = PAGE, listScroll = 0, lastKey = '', lastHref = null, seq = 0, artCat = null, virtual = null;

    // всё рисуем через textContent — текст новостей не может встроить свой HTML
    function el(tag, cls, text) {
      var e = document.createElement(tag);
      if (cls) e.className = cls;
      if (text != null) e.textContent = text;
      return e;
    }

    /* ---------- адреса ---------- */
    function findCat(v) {
      if (!v) return null;
      for (var i = 0; i < CATS.length; i++) if (CATS[i].slug === v || CATS[i].label === v) return CATS[i];
      return null;
    }
    function urlNews(id) {
      if (CONFIG.urlMode === 'hash') return '#n/' + id;
      if (CONFIG.urlMode === 'path') return CONFIG.pathBase + id;
      return location.pathname + '?news=' + id;
    }
    function urlCat(c) {
      if (CONFIG.urlMode === 'hash') return c ? '#c/' + c.slug : '#';
      if (CONFIG.urlMode === 'path') return CONFIG.pathBase + (c ? 'c/' + c.slug : '');
      return location.pathname + (c ? '?cat=' + c.slug : '');
    }
    function abs(u) {
      try { return new URL(u, location.href.split(/[?#]/)[0]).href.replace(/#$/, ''); }
      catch (e) { return u; }
    }
    function dec(s) { try { return decodeURIComponent(s); } catch (e) { return s; } }

    // понимаем все виды адресов сразу (и старые ссылки с #), независимо от urlMode
    function route() {
      var q = new URLSearchParams(location.search), h = dec((virtual !== null ? virtual : location.hash).slice(1));
      var news = q.get('news'), cat = q.get('cat');
      var p = location.pathname, b = CONFIG.pathBase;
      if (CONFIG.urlMode === 'path' && p.toLowerCase().indexOf(b.toLowerCase()) === 0) {
        var rest = dec(p.slice(b.length)).replace(/\/+$/, '');
        if (rest.indexOf('c/') === 0) cat = rest.slice(2); else if (rest) news = rest;
      }
      if (!news && h.indexOf('n/') === 0) news = h.slice(2);
      if (!news && !cat && h.indexOf('c/') === 0) cat = h.slice(2);
      if (news) return {page: 'news', id: news};
      return {page: 'list', cat: findCat(cat)};
    }

    /* ---------- SEO: title, description, canonical, Open Graph, микроразметка ---------- */
    var tags = {}, origTitle = null;
    function tag(key, selector, make, attr, value) {
      var t = tags[key];
      if (!t) {
        var e = document.head.querySelector(selector), created = !e;
        if (created) { e = make(); document.head.appendChild(e); }
        t = tags[key] = {el: e, created: created, attr: attr, orig: created ? null : e.getAttribute(attr)};
      }
      t.el.setAttribute(attr, value);
    }
    function meta(name, value, prop) {
      var a = prop ? 'property' : 'name';
      tag(a + ':' + name, 'meta[' + a + '="' + name + '"]', function () {
        var m = document.createElement('meta'); m.setAttribute(a, name); return m;
      }, 'content', value);
    }
    function dropTag(key) {   // вернуть как было до нас
      var t = tags[key];
      if (!t) return;
      if (t.created) t.el.remove(); else if (t.orig != null) t.el.setAttribute(t.attr, t.orig);
      delete tags[key];
    }
    function dropLd() { var s = document.getElementById('mbn-ld'); if (s) s.remove(); }

    function seo(o) {   // ошибка в SEO-части не должна ломать показ новостей
      try { seoApply(o); } catch (e) {}
    }
    function seoApply(o) {
      if (origTitle === null) origTitle = document.title;
      document.title = o.title;
      meta('description', o.desc);
      tag('canonical', 'link[rel="canonical"]', function () {
        var l = document.createElement('link'); l.rel = 'canonical'; return l;
      }, 'href', o.url);
      meta('og:title', o.title, true);
      meta('og:description', o.desc, true);
      meta('og:url', o.url, true);
      meta('og:type', o.type, true);
      meta('og:site_name', CONFIG.siteName, true);
      if (o.image) meta('og:image', o.image, true); else dropTag('property:og:image');
      if (o.notFound) { meta('robots', 'noindex'); meta('prerender-status-code', '404'); }
      else { dropTag('name:robots'); dropTag('name:prerender-status-code'); }
      dropLd();
      if (o.ld) {
        var s = document.createElement('script');
        s.type = 'application/ld+json'; s.id = 'mbn-ld';
        s.textContent = JSON.stringify(o.ld).replace(/</g, '\\u003c');
        document.head.appendChild(s);
      }
    }
    function crumbs(list) {
      return {'@type': 'BreadcrumbList', itemListElement: list.map(function (c, i) {
        return {'@type': 'ListItem', position: i + 1, name: c[0], item: c[1]};
      })};
    }
    function ready() { window.prerenderReady = true; }

    /* ---------- кусочки вёрстки ---------- */
    function when(iso, full) {
      var d = new Date(iso);
      var t = d.toLocaleTimeString('ru-RU', {hour: '2-digit', minute: '2-digit'});
      if (full) return d.toLocaleDateString('ru-RU', {day: 'numeric', month: 'long', year: 'numeric'}) + ', ' + t;
      var a = new Date(d), b = new Date();
      a.setHours(0, 0, 0, 0); b.setHours(0, 0, 0, 0);
      var days = Math.round((b - a) / 864e5);
      if (days === 0) return 'Сегодня, ' + t;
      if (days === 1) return 'Вчера, ' + t;
      return d.toLocaleDateString('ru-RU', {day: 'numeric', month: 'long'}) + ', ' + t;
    }

    function link(cls, href, text) {   // внутренняя ссылка: настоящий href для поисковиков, переход без перезагрузки
      var a = el('a', cls, text);
      a.href = href;
      a.setAttribute('data-nav', '');
      return a;
    }

    function img(src, alt, eager) {
      var i = el('img', 'img');
      i.alt = alt || '';
      i.loading = eager ? 'eager' : 'lazy';
      i.width = 3; i.height = 2;       // пропорции, чтобы страница не прыгала при загрузке
      if (src) i.src = BASE + src;
      return i;
    }

    function metaLine(n, full) {
      var m = el('div', 'meta');
      var c = findCat(n.category);
      if (full && c) m.appendChild(link('cat', urlCat(c), n.category)); else m.appendChild(el('span', 'cat', n.category));
      m.appendChild(document.createTextNode(' · '));
      var t = el('time', null, when(n.published, full));
      t.setAttribute('datetime', n.published);
      m.appendChild(t);
      if (full) m.appendChild(document.createTextNode(' · ' + CONFIG.author));
      return m;
    }

    function card(n, big) {
      var a = link(big ? 'hero' : 'row', urlNews(n.id));
      a.appendChild(img(big ? n.image : (n.thumb || n.image), n.title, big));
      var d = el('div');
      d.appendChild(metaLine(n));
      d.appendChild(el(big ? 'h2' : 'h3', null, n.title));
      d.appendChild(el('p', null, n.lead));
      a.appendChild(d);
      return a;
    }

    function drawHeader(isHome, cat) {
      // логотип — h1 только на главной ленты: на странице всегда ровно один h1
      logo.innerHTML = '';
      var box = el(isHome ? 'h1' : 'div', 'logo');
      var a = link(null, urlCat(null));
      a.appendChild(document.createTextNode('Новости '));
      a.appendChild(el('span', null, 'спорта'));
      box.appendChild(a);
      logo.appendChild(box);

      nav.innerHTML = '';
      var present = {};
      items.forEach(function (n) { present[n.category] = 1; });
      [null].concat(CATS.filter(function (c) { return present[c.label]; })).forEach(function (c) {
        var li = el('li');
        li.appendChild(link(c === cat || (c && cat && c.slug === cat.slug) ? 'on' : '', urlCat(c), c ? c.label : 'Все новости'));
        nav.appendChild(li);
      });
    }

    /* ---------- лента ---------- */
    function drawList(cat) {
      drawHeader(!cat, cat);
      var list = items.filter(function (n) { return !cat || n.category === cat.label; });
      view.innerHTML = '';
      if (cat) view.appendChild(el('h1', 'h1c', cat.h1));
      if (!list.length) view.appendChild(el('div', 'empty', loaded ? 'Новостей пока нет' : 'Загружаем новости…'));
      else {
        view.appendChild(card(list[0], true));
        list.slice(1, shown).forEach(function (n) { view.appendChild(card(n)); });
        if (list.length > shown) {
          var b = el('button', 'more', 'Показать ещё');
          b.type = 'button';
          b.onclick = function () { shown += PAGE; drawList(cat); };
          view.appendChild(b);
        }
      }
      var url = abs(urlCat(cat));
      seo({
        title: (cat ? cat.h1 : HOME_TITLE) + ' — ' + CONFIG.siteName,
        desc: cat ? cat.h1 + ': свежие события, результаты, трансферы и заявления.' : HOME_DESC,
        url: url, type: 'website',
        image: list.length && list[0].image ? BASE + list[0].image : null,
        ld: cat ? {'@context': 'https://schema.org', '@graph': [crumbs([[HOME_TITLE, abs(urlCat(null))], [cat.h1, url]])]} : null
      });
      if (loaded) ready();
    }

    /* ---------- страница новости ---------- */
    function fillOthers(box, id) {
      var old = box.querySelector('.also');
      if (old) old.remove();
      var others = items.filter(function (x) { return x.id !== id; }).slice(0, 5);
      if (!others.length) return;
      var also = el('aside', 'also');
      also.appendChild(el('div', 'also-t', 'Другие новости'));
      others.forEach(function (x) { also.appendChild(card(x)); });
      box.appendChild(also);
    }

    function drawNews(id) {
      var my = ++seq;
      artCat = null;
      drawHeader(false, null);
      view.innerHTML = '';
      var box = el('article', 'art');
      box.appendChild(link('back', urlCat(null), '← Все новости'));
      var note = el('div', 'empty', 'Загружаем…');
      box.appendChild(note);
      view.appendChild(box);

      function fail(notFound) {
        if (my !== seq) return;
        note.textContent = notFound ? 'Новость не найдена. Возможно, она устарела и удалена из архива.' : 'Не удалось загрузить новость. Обновите страницу.';
        if (notFound) seo({title: 'Новость не найдена — ' + CONFIG.siteName, desc: HOME_DESC, url: abs(urlCat(null)), type: 'website', notFound: true});
        ready();
      }
      if (!/^[a-z0-9-]+$/i.test(id)) { fail(true); return; }

      fetch(BASE + 'a/' + id + '.json')
        .then(function (r) {
          if (r.status === 404) { fail(true); return null; }
          if (!r.ok) throw new Error(r.status);
          return r.json();
        })
        .then(function (n) {
          if (!n || my !== seq) return;
          var cat = artCat = findCat(n.category);
          drawHeader(false, cat);
          box.removeChild(note);
          box.appendChild(metaLine(n, true));
          box.appendChild(el('h1', null, n.title));
          if (n.lead) box.appendChild(el('p', 'lead', n.lead));
          if (n.image) { var f = el('figure'); f.appendChild(img(n.image, n.title, true)); box.appendChild(f); }
          var body = el('div', 'body');
          (n.body || []).forEach(function (p) { body.appendChild(el('p', null, p)); });
          box.appendChild(body);
          fillOthers(box, n.id);

          var url = abs(urlNews(n.id)), home = abs(urlCat(null));
          var trail = [[HOME_TITLE, home]];
          if (cat) trail.push([cat.h1, abs(urlCat(cat))]);
          trail.push([n.title, url]);
          var article = {
            '@type': 'NewsArticle', headline: n.title.slice(0, 110), description: n.lead,
            datePublished: n.published, dateModified: n.modified || n.published,
            articleSection: n.category, inLanguage: 'ru',
            mainEntityOfPage: {'@type': 'WebPage', '@id': url},
            author: {'@type': 'Organization', name: CONFIG.author},
            publisher: {'@type': 'Organization', name: CONFIG.siteName}
          };
          if (n.image) article.image = [BASE + n.image];
          seo({
            title: n.title + ' — ' + CONFIG.siteName, desc: n.lead || n.title, url: url, type: 'article',
            image: n.image ? BASE + n.image : null,
            ld: {'@context': 'https://schema.org', '@graph': [article, crumbs(trail)]}
          });
          ready();
        })
        .catch(function () { fail(false); });
    }

    /* ---------- переходы ---------- */
    function render(isNav) {
      var r = route();
      var key = r.page + '/' + (r.id || (r.cat ? r.cat.slug : ''));
      if (isNav && lastKey.indexOf('list/') === 0) listScroll = window.scrollY;
      if (r.page === 'news') {
        drawNews(r.id);
        if (isNav) root.scrollIntoView();
      } else {
        seq++;
        if (key !== lastKey && lastKey.indexOf('list/') === 0) shown = PAGE;   // сменили раздел
        drawList(r.cat);
        if (isNav) {
          if (lastKey.indexOf('news/') === 0 && listScroll) window.scrollTo(0, listScroll);
          else root.scrollIntoView();
        }
      }
      lastKey = key;
      lastHref = location.href;
    }

    function onUrlChange() { if (location.href !== lastHref) { virtual = null; render(true); } }

    function onClick(e) {
      if (e.defaultPrevented || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
      var a = e.target.closest ? e.target.closest('a[data-nav]') : null;
      if (!a || !root.contains(a)) return;
      e.preventDefault();
      if (CONFIG.urlMode === 'hash') {
        var target = a.getAttribute('href'), before = location.href;
        try { location.hash = target; } catch (err) {}
        // окружение не даёт менять адрес (например, предпросмотр файла) — переходим «внутри» страницы
        if (location.href === before) { virtual = target; render(true); }
        return;
      }
      try { history.pushState(null, '', a.getAttribute('href')); }
      catch (err) { location.href = a.href; return; }
      onUrlChange();
    }

    /* ---------- загрузка ленты ---------- */
    function load() {
      fetch(BASE + 'news.json?t=' + Date.now())
        .then(function (r) { if (!r.ok) throw new Error(r.status); return r.json(); })
        .then(function (data) {
          loaded = true;
          items = (data.items || []).filter(function (n) { return n.category; });
          upd.className = 'foot';
          upd.textContent = data.updated_at ? 'Лента обновлена: ' + when(data.updated_at) : '';
          var r = route();
          if (r.page === 'list') drawList(r.cat);
          else {   // на странице статьи экран не дёргаем — только обновляем разделы и «Другие новости»
            drawHeader(false, artCat);
            var box = view.querySelector('.art');
            if (box && box.querySelector('.body')) { fillOthers(box, r.id); }
          }
        })
        .catch(function (e) {
          upd.className = 'foot err';
          upd.textContent = 'Не удалось загрузить новости (' + e.message + ')';
          if (!loaded && route().page === 'list') { view.innerHTML = ''; ready(); }
        });
    }

    root.addEventListener('click', onClick);
    window.addEventListener('popstate', onUrlChange);
    window.addEventListener('hashchange', onUrlChange);
    var timer = setInterval(load, 120000);
    // ушли с лендинга на другую страницу сайта — убираем за собой title, canonical и микроразметку
    var guard = setInterval(function () { if (!document.documentElement.contains(root)) window.__mbnStop(); }, 1500);

    window.__mbnStop = function () {
      clearInterval(timer); clearInterval(guard);
      root.removeEventListener('click', onClick);
      window.removeEventListener('popstate', onUrlChange);
      window.removeEventListener('hashchange', onUrlChange);
      Object.keys(tags).forEach(dropTag);
      dropLd();
      if (origTitle !== null) document.title = origTitle;
      window.__mbnStop = null;
    };

    render(false);
    load();
  })();
