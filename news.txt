    <div class="top">

  <script>
  (function () {
    var BASE = 'https://raw.githubusercontent.com/anastasiyamakarenko/melbet-news/main/data/';
    var CATS = ['Футбол', 'Хоккей', 'Баскетбол', 'Теннис', 'Фигурное катание', 'Волейбол', 'Биатлон',
                'Единоборства', 'Автоспорт', 'Киберспорт', 'Другое'];
    var PAGE = 20;

    var root = document.getElementById('mbn');
    var nav = document.getElementById('mbn-nav');
    var view = document.getElementById('mbn-view');
    var upd = document.getElementById('mbn-upd');
    var items = [], shown = PAGE, listScroll = 0, lastRoute = '';

    // всё рисуем через textContent — текст из RSS не может встроить свой HTML
    function el(tag, cls, text) {
      var e = document.createElement(tag);
      if (cls) e.className = cls;
      if (text != null) e.textContent = text;
      return e;
    }

    function when(iso, full) {
      var d = new Date(iso);
      var t = d.toLocaleTimeString('ru-RU', {hour: '2-digit', minute: '2-digit'});
      if (full) return d.toLocaleDateString('ru-RU', {day: 'numeric', month: 'long', year: 'numeric'}) + ', ' + t;
      var a = new Date(d), b = new Date();
      a.setHours(0, 0, 0, 0); b.setHours(0, 0, 0, 0);
      var days = Math.round((b - a) / 864e5);
      if (days === 0) return 'Сегодня, ' + t;
      if (days === 1) return 'Вчера, ' + t;
      return d.toLocaleDateString('ru-RU', {day: 'numeric', month: 'long'}) + ', ' + t;
    }

    function img(src) {
      var i = el('img', 'img');
      i.alt = '';
      i.loading = 'lazy';
      if (src) i.src = BASE + src;
      return i;
    }

    function meta(n, full) {
      var m = el('div', 'meta');
      m.appendChild(el('span', 'cat', n.category));
      m.appendChild(document.createTextNode(' · ' + when(n.published, full)));
      return m;
    }

    function link(n) { return '#n/' + n.id; }

    function hero(n) {
      var a = el('a', 'hero'); a.href = link(n);
      a.appendChild(img(n.image));
      var d = el('div');
      d.appendChild(meta(n));
      d.appendChild(el('h2', null, n.title));
      d.appendChild(el('p', null, n.lead));
      a.appendChild(d);
      return a;
    }

    function row(n) {
      var a = el('a', 'row'); a.href = link(n);
      a.appendChild(img(n.thumb || n.image));
      var d = el('div');
      d.appendChild(meta(n));
      d.appendChild(el('h3', null, n.title));
      d.appendChild(el('p', null, n.lead));
      a.appendChild(d);
      return a;
    }

    function route() {
      var h = decodeURIComponent(location.hash.slice(1));
      if (h.indexOf('n/') === 0) return {page: 'news', id: h.slice(2)};
      if (h.indexOf('c/') === 0) return {page: 'list', cat: h.slice(2)};
      return {page: 'list', cat: ''};
    }

    function drawNav(cat) {
      nav.innerHTML = '';
      var present = {};
      items.forEach(function (n) { present[n.category] = 1; });
      [''].concat(CATS.filter(function (c) { return present[c]; })).forEach(function (c) {
        var a = el('a', c === cat ? 'on' : '', c || 'Все новости');
        a.href = c ? '#c/' + encodeURIComponent(c) : '#';
        nav.appendChild(a);
      });
    }

    function drawList(cat) {
      drawNav(cat);
      var list = items.filter(function (n) { return !cat || n.category === cat; });
      view.innerHTML = '';
      if (!list.length) { view.appendChild(el('div', 'empty', 'Новостей пока нет')); return; }
      view.appendChild(hero(list[0]));
      list.slice(1, shown).forEach(function (n) { view.appendChild(row(n)); });
      if (list.length > shown) {
        var b = el('button', 'more', 'Показать ещё');
        b.onclick = function () { shown += PAGE; drawList(cat); };
        view.appendChild(b);
      }
    }

    function drawNews(id) {
      var n0 = items.filter(function (n) { return n.id === id; })[0];
      drawNav(n0 ? n0.category : '');
      view.innerHTML = '';
      var box = el('div', 'art');
      var back = el('a', 'back', '← Все новости'); back.href = '#';
      box.appendChild(back);
      box.appendChild(el('div', 'empty', 'Загружаем…'));
      view.appendChild(box);

      fetch(BASE + 'a/' + id + '.json')
        .then(function (r) { if (!r.ok) throw new Error(r.status); return r.json(); })
        .then(function (n) {
          box.removeChild(box.lastChild);
          box.appendChild(meta(n, true));
          box.appendChild(el('h1', null, n.title));
          if (n.lead) box.appendChild(el('p', 'lead', n.lead));
          if (n.image) box.appendChild(img(n.image));
          var body = el('div', 'body');
          (n.body || []).forEach(function (p) { body.appendChild(el('p', null, p)); });
          box.appendChild(body);

          var others = items.filter(function (x) { return x.id !== n.id; }).slice(0, 5);
          if (others.length) {
            var also = el('div', 'also');
            also.appendChild(el('div', 'also-t', 'Другие новости'));
            others.forEach(function (x) { also.appendChild(row(x)); });
            box.appendChild(also);
          }
        })
        .catch(function () {
          box.lastChild.textContent = 'Эта новость больше недоступна';
        });
    }

    function render(fromHashChange) {
      var r = route();
      var key = r.page + '/' + (r.id || r.cat);
      if (fromHashChange && lastRoute.indexOf('list/') === 0) listScroll = window.scrollY;
      if (r.page === 'news') {
        drawNews(r.id);
        if (fromHashChange) root.scrollIntoView();
      } else {
        if (key !== lastRoute && lastRoute.indexOf('list/') === 0) shown = PAGE;  // сменили раздел
        drawList(r.cat);
        if (fromHashChange) {
          if (lastRoute.indexOf('news/') === 0 && listScroll) window.scrollTo(0, listScroll);
          else root.scrollIntoView();
        }
      }
      lastRoute = key;
    }

    function load() {
      fetch(BASE + 'news.json?t=' + Date.now())
        .then(function (r) { if (!r.ok) throw new Error(r.status); return r.json(); })
        .then(function (data) {
          var first = !items.length;
          items = (data.items || []).filter(function (n) { return n.category; });
          upd.className = 'foot';
          upd.textContent = data.updated_at ? 'Лента обновлена: ' + when(data.updated_at) : '';
          // на странице статьи не дёргаем экран — свежие новости появятся в ленте при возврате
          if (first || route().page === 'list') render(false);
        })
        .catch(function (e) {
          if (!items.length) view.innerHTML = '';
          upd.className = 'foot err';
          upd.textContent = 'Не удалось загрузить новости (' + e.message + ')';
        });
    }

    window.addEventListener('hashchange', function () { render(true); });
    load();
    setInterval(load, 120000);
  })();
  </script>
</div>
