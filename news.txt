  <div class="mbn-wrap">
    <h1>Новости дня</h1>
    <div class="mbn-upd" id="mbn-upd">Загружаем новости…</div>
    <div id="mbn-list"></div>
  </div>
  <script>
  (function () {
    var BASE = 'https://raw.githubusercontent.com/anastasiyamakarenko/melbet-news/main/data/';
    var list = document.getElementById('mbn-list');
    var upd = document.getElementById('mbn-upd');

    function render(data) {
      list.innerHTML = '';
      (data.items || []).forEach(function (n) {
        var a = document.createElement('article');
        if (n.image) {
          var img = document.createElement('img');
          img.src = BASE + n.image;
          img.alt = '';
          img.loading = 'lazy';
          a.appendChild(img);
        }
        var h = document.createElement('h2'); h.textContent = n.title; a.appendChild(h);
        var p = document.createElement('p'); p.textContent = n.text; a.appendChild(p);
        list.appendChild(a);
      });
      upd.textContent = data.updated_at
        ? 'Обновлено: ' + new Date(data.updated_at).toLocaleString('ru-RU', {day:'numeric', month:'long', hour:'2-digit', minute:'2-digit'})
        : 'Ждём первое обновление';
    }

    function load() {
      fetch(BASE + 'news.json?t=' + Date.now())
        .then(function (r) { if (!r.ok) throw new Error(r.status); return r.json(); })
        .then(render)
        .catch(function (e) {
          upd.textContent = 'Не удалось загрузить новости (' + e.message + ')';
          upd.className = 'mbn-upd mbn-err';
        });
    }

    load();
    setInterval(load, 120000);
  })();
  </script>
